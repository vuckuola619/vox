async function callClaude(system, user, maxTokens){
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method:"POST",
    headers:{"Content-Type":"application/json"},
    body: JSON.stringify({
      model:"claude-sonnet-4-6",
      max_tokens: maxTokens || 1000,
      system: system,
      messages:[{role:"user", content:user}]
    })
  });
  const data = await res.json();
  const text = (data.content||[]).filter(b=>b.type==="text").map(b=>b.text).join("\n");
  return text;
}
function extractBalancedJson(text){
  let startIdx = -1, openChar = null, closeChar = null;
  for(let i=0;i<text.length;i++){
    if(text[i] === '{'){ startIdx = i; openChar='{'; closeChar='}'; break; }
    if(text[i] === '['){ startIdx = i; openChar='['; closeChar=']'; break; }
  }
  if(startIdx === -1) throw new Error('No JSON object found in response');
  let depth = 0, inStr = false, esc = false;
  for(let i=startIdx; i<text.length; i++){
    const ch = text[i];
    if(inStr){
      if(esc) esc = false;
      else if(ch === '\\') esc = true;
      else if(ch === '"') inStr = false;
    } else {
      if(ch === '"') inStr = true;
      else if(ch === openChar) depth++;
      else if(ch === closeChar){ depth--; if(depth === 0) return text.slice(startIdx, i+1); }
    }
  }
  throw new Error('Response was cut off before the JSON closed (hit the token limit) - try a smaller batch or try again');
}
function repairStrayQuotes(jsonStr){
  // Walk the JSON string and escape any " that appears where it can't legally close/open a token,
  // i.e. a quote inside a string value that isn't followed by a JSON structural character.
  let out = '';
  let inStr = false, esc = false;
  for(let i=0;i<jsonStr.length;i++){
    const ch = jsonStr[i];
    if(!inStr){
      out += ch;
      if(ch === '"') inStr = true;
      continue;
    }
    // inStr === true
    if(esc){ out += ch; esc = false; continue; }
    if(ch === '\\'){ out += ch; esc = true; continue; }
    if(ch === '"'){
      const rest = jsonStr.slice(i+1).match(/^\s*/)[0];
      const next = jsonStr[i+1+rest.length];
      if(next === undefined || ':,}]'.includes(next)){
        out += ch; inStr = false;
      } else {
        out += '\\"';
      }
      continue;
    }
    out += ch;
  }
  return out;
}
function parseJSON(text){
  let clean = text.trim();
  clean = clean.replace(/^```json/i,'').replace(/^```/,'').replace(/```$/,'').trim();
  const span = extractBalancedJson(clean);
  try{
    return JSON.parse(span);
  }catch(e){
    return JSON.parse(repairStrayQuotes(span));
  }
}

/* ---------------- beat splitting (deterministic, no API) ---------------- */
function splitBeats(script){
  const sentences = script.match(/[^.!?]+[.!?]+/g) || [script];
  let beats = [];
  sentences.forEach(raw=>{
    const s = raw.trim();
    if(!s) return;
    const words = s.split(/\s+/);
    if(words.length <= 8){ beats.push(s); return; }
    const parts = s.split(/,\s*/).map(p=>p.trim()).filter(Boolean);
    if(parts.length <= 1){
      for(let i=0;i<words.length;i+=7) beats.push(words.slice(i,i+7).join(' '));
      return;
    }
    let buffer = [];
    parts.forEach((part,idx)=>{
      const w = part.split(/\s+/);
      if(buffer.length === 0){ buffer = w; }
      else if(buffer.length + w.length <= 8){ buffer = buffer.concat(w); }
      else { beats.push(buffer.join(' ') + (idx < parts.length ? ',' : '')); buffer = w; }
    });
    if(buffer.length) beats.push(buffer.join(' '));
  });
  let cum = 0;
  return beats.map(b=>{
    const start = +(cum/2.5).toFixed(1);
    cum += wc(b);
    return {text:b, start};
  });
}

    const before = val.slice(0, cursor);
